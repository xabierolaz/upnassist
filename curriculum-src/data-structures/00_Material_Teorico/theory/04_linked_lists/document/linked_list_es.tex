\section{¿Qué es una Lista Enlazada?}

	\begin{frame}
		\frametitle{Visión General de una Lista Enlazada}
		\begin{itemize}
			\item Una lista es una colección de elementos
			\item En una \textbf{Lista Enlazada} cada elemento de la lista apuntará al siguiente elemento
			\item Cada elemento se llama \textbf{Nodo}
			\item Un Nodo contendrá dos variables
			\begin{itemize}
				\item Los datos que queremos almacenar
				\item La \textit{referencia} al siguiente Nodo
			\end{itemize}
		\end{itemize}
		\vspace{1cm}
		\begin{center}
			\begin{tikzpicture}
				\node[list,on chain] (A) {12};
				\node[list,on chain] (B) {99};
				\node[list,on chain] (C) {37};
				\node[squarecross]   (D) [right=of C] {};
				\draw[chain_arrow] let \p1 = (A.two), \p2 = (A.center) in (\x1,\y2) -- (B);
				\draw[chain_arrow] let \p1 = (B.two), \p2 = (B.center) in (\x1,\y2) -- (C);
				\draw[chain_arrow] let \p1 = (C.two), \p2 = (C.center) in (\x1,\y2) -- (D);
			\end{tikzpicture}
		\end{center}
	\end{frame}

	\begin{frame}[fragile]
		\frametitle{Cómo crear un Nodo}
		\begin{itemize}
			\item Un Nodo contendrá dos puntos de datos
			\begin{itemize}
				\item Los datos que queremos almacenar
				\item La referencia al siguiente elemento
			\end{itemize}
			\item Podemos usar una lista simple para almacenarlo \lstinline|[dato, siguiente]|
			\item Pero es más fácil usar una clase
		\end{itemize}
\pause
\begin{lstlisting}
class Nodo:
	def __init__(self, dato, siguiente = None):
		self.dato = dato
		self.siguiente = siguiente
\end{lstlisting}
	\end{frame}

	\begin{frame}[fragile]
		\frametitle{Añadiendo Relaciones a los Nodos}
		\begin{itemize}
			\item Una vez que hemos creado un Nodo para almacenar nuestros datos podemos empezar a encadenarlos
			\begin{itemize}
				\item De la misma manera que funcionan los eslabones de una cadena
			\end{itemize}
			\item Podemos crear nodos y unirlos mediante el atributo siguiente (next)
		\end{itemize}
		\vspace{0.5cm}
  		\begin{center}
  			\begin{tikzpicture}
  				\node[list,on chain] (A) {\lstinline|"aaaa"|};
  				\node[list,on chain] (B) {\lstinline|"bbbb"|};
  				\node[squarecross]   (D) [right=of B] {};
  				\draw[chain_arrow] let \p1 = (A.two), \p2 = (A.center) in (\x1,\y2) -- (B);
  				\draw[chain_arrow] let \p1 = (B.two), \p2 = (B.center) in (\x1,\y2) -- (D);
  			\end{tikzpicture}
  		\end{center}
        \vspace{0.2cm}
        \pause
\begin{lstlisting}
primero = Nodo("aaaa")
segundo = Nodo("bbbb")

primero.siguiente = segundo
\end{lstlisting}
		
	\end{frame}

	\begin{frame}
		\frametitle{Construyendo la Lista Enlazada}
		\begin{itemize}
			\item Hemos creado Nodos y los hemos enlazado definiendo el siguiente nodo
			\item Pero tener que cambiar el valor del atributo siguiente del último elemento de la lista es engorroso
			\item El siguiente paso es proporcionar una interfaz similar a las listas en Python
			\begin{itemize}
				\item Insertar un elemento (al principio, al final o en medio)
				\item Eliminar un elemento
				\item Obtener un elemento
				\item Obtener el tamaño de la lista
				\item Comprobar si un elemento existe en la lista
				\item ...
			\end{itemize}
		\end{itemize}
		\vspace{0.7cm}
		\begin{center}
			\begin{tikzpicture}
				\node[list,on chain] (A) {cabeza};
				\node[list,on chain] (B) {medio};
				\node[list,on chain] (C) {cola};
				\node[squarecross]   (D) [right=of C] {};
				\draw[chain_arrow] let \p1 = (A.two), \p2 = (A.center) in (\x1,\y2) -- (B);
				\draw[chain_arrow] let \p1 = (B.two), \p2 = (B.center) in (\x1,\y2) -- (C);
				\draw[chain_arrow] let \p1 = (C.two), \p2 = (C.center) in (\x1,\y2) -- (D);
			\end{tikzpicture}
		\end{center}
	\end{frame}

	\begin{frame}[fragile]
		\frametitle{Usando una Clase para la Lista Enlazada}
		\begin{itemize}
		\item Podemos crear una clase que maneje las operaciones por nosotros
		\item La clase tendrá un atributo que referencia la cabeza de la lista
			\begin{itemize}
				\item Se puede llegar a los siguientes elementos saltando de un nodo a otro
				\item Siempre empieza en la cabeza
			\end{itemize}
		\item Trabajaremos con la lista enlazada, y no con nodos individuales
		\end{itemize}
\begin{lstlisting}
class ListaEnlazada:
	def __init__(self, cabeza = None):
		self.__cabeza = cabeza
\end{lstlisting}
	\end{frame}

\section{Funciones Básicas}

	\begin{frame}[fragile]
		\frametitle{Obteniendo el Tamaño}
		\begin{itemize}
			\item \lstinline|lista_enlazada.size()|
			\item Empezando por la cabeza \textbf{recorremos} la lista hasta llegar al final
			\begin{itemize}
				\item Moverse de un Nodo a otro usando \texttt{siguiente}
			\end{itemize}
			\item En cada salto añadimos uno a un contador
		\end{itemize}
		\vspace{0.5cm}
        \pause
		\begin{lstlisting}
def size(self) -> int:
	actual = self.__cabeza
	contador = 0
	while actual is not None:
		contador += 1
		actual = actual.siguiente
	return contador
		\end{lstlisting}
	\end{frame}

	\begin{frame}[fragile]
		\frametitle{Comprobando si un Elemento está en la Lista}
		\begin{itemize}
			\item \lstinline|lista_enlazada.contains(elemento)|
			\item Empezando por la cabeza recorremos la lista hasta llegar al final
			\begin{itemize}
				\item Moverse de un Nodo a otro usando \texttt{siguiente}
			\end{itemize}
			\item En cada salto probamos si el elemento en el Nodo es el que estamos buscando
			\item Si no lo encontramos, devolvemos \lstinline|False|
		\end{itemize}
		\vspace{0.5cm}
        \pause
		\begin{lstlisting}
def contains(self, item) -> bool:
	actual = self.__cabeza
	while actual is not None:
		if actual.dato == item:
			return True
		actual = actual.siguiente
	return False
		\end{lstlisting}
	\end{frame}

	\begin{frame}[fragile]
		\frametitle{Obteniendo un Elemento por Índice}
		\begin{itemize}
			\item \lstinline|lista_enlazada.get(indice)|
			\item El método devolverá el elemento en un índice específico
			\item Tendremos que usar un contador para comprobar si hemos llegado al elemento deseado
		\end{itemize}
		\vspace{0.5cm}
        \pause
\begin{lstlisting}
def get(self, indice: int):
	actual = self.__cabeza
	for i in range(indice):
		actual = actual.siguiente
	return actual.dato
\end{lstlisting}
	\end{frame}

	\begin{frame}[fragile]
	\frametitle{Establecer un Elemento en un Índice}
	\begin{itemize}
		\item \lstinline|lista_enlazada.set(indice, item)|
		\item El método cambiará el valor de un elemento en un índice dado
		\item Al igual que con el anterior, tendremos que usar un contador para comprobar si hemos llegado al elemento deseado
	\end{itemize}
	\vspace{0.5cm}
    \pause
\begin{lstlisting}
def set(self, indice: int, item):
	actual = self.__cabeza
	for i in range(indice):
		actual = actual.siguiente
	actual.dato = item
\end{lstlisting}
\end{frame}

def add_tail(self, item):
	if self.__cabeza is None:
		self.__cabeza = Nodo(item)
	else:
		actual = self.__cabeza
		while actual.siguiente is not None:
			actual = actual.siguiente
		actual.siguiente = Nodo(item)
\end{lstlisting}
\end{frame}

\section{Eliminando Elementos}

	\begin{frame}
		\frametitle{Eliminando un Elemento}
		\begin{itemize}
			\item Queremos eliminar un Nodo de la lista
			\begin{itemize}
				\item En la cabeza
				\item Al final
				\item En un índice
			\end{itemize}
			\item Tendremos que dejar de referenciar el Nodo que queremos borrar
			\begin{itemize}
				\item Hacer el nodo inalcanzable para la cadena
				\item Siempre seguimos la cadena empezando por la cabeza y siguiendo las flechas
			\end{itemize}
		\end{itemize}

	\begin{frame}
		\frametitle{Eliminando la Cabeza}
		\begin{itemize}
			\item \lstinline|lista_enlazada.remove_head()|
			\item El atributo \texttt{\___}cabeza\texttt{\___} de la lista enlazada dejará de apuntar al primer elemento
			\begin{itemize}
				\item Establecemos la nueva \texttt{\___}cabeza\texttt{\___} al valor del atributo \texttt{siguiente} de la antigua
				\item La antigua cabeza será inalcanzable
			\end{itemize}
		\end{itemize}
		\vspace{0.7cm}
		\begin{center}

def remove_tail(self):
	if self.__cabeza is None:
		return None
	if self.__cabeza.siguiente is None:
		self.__cabeza = None
	else:
		actual = self.__cabeza
		while actual.siguiente.siguiente is not None:
			actual = actual.siguiente
		actual.siguiente = None

	\begin{frame}

		\begin{itemize}

			\item Mantener un contador en la clase para saber el número de elementos, de esta forma no tenemos que recorrer la lista para saber el tamaño
