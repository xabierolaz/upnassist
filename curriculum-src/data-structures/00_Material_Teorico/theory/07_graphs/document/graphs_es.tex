\section{¿Qué es un Grafo?}

\begin{frame}\frametitle{¿Qué es un Grafo?}
\begin{itemize}
\item ¿Cómo puede Google Maps encontrar el camino de una ciudad a otra?
\item ¿Cómo Facebook, Twitter... rastrean las relaciones entre usuarios?
\item Si intentamos representar Internet como un dibujo, ¿qué forma tendría?
\end{itemize}
\end{frame}

\begin{frame}\frametitle{Definición}
\begin{itemize}
\item Un grafo es una estructura que representa relaciones entre elementos
\item Una tupla de vértices y aristas
\begin{itemize}
    \item Un conjunto de vértices: $V = {v_1, v_2, ..., v_n}$
    \item Un conjunto de aristas: $E = {e_1, e_2, ..., e_m}$
\end{itemize}
\pause
\hspace{1cm}
\begin{block}{}
$$
G = (V, E)
$$
\end{block}
\end{itemize}
\end{frame}

\section{Representación de Grafos}

\begin{frame}\frametitle{Del Concepto a la Estructura de Datos}
\begin{itemize}
\item Un grafo es útil para las siguientes preguntas:
    \begin{itemize}
    \item ¿Cuál es el camino más corto entre dos ciudades?
    \item ¿Hay una ruta que visite cada nodo exactamente una vez? 
    \item ¿Cuántos grados de separación hay entre dos usuarios?
    \item ¿Hay comunidades o grupos de personas?
    \item ¿Qué páginas web son más importantes? (PageRank)
    \end{itemize}
\item Necesitamos una Estructura de Datos para almacenar la información del grafo y trabajar con ella
\item Hay diferentes enfoques
    \begin{itemize}
    \item Según la información en el grafo (denso o disperso)
    \item Según las consultas comunes (qué aristas tiene un vértice o si existe la arista (u, v))
    \end{itemize}
\end{itemize}
\end{frame}

\subsection{Cuál elegir}

\begin{frame}\frametitle{Complejidad}
\begin{center}
\begin{tabular}{@{} lll @{}}
\toprule
\textbf{Operación} & \textbf{Matriz Adyacencia} & \textbf{Lista Adyacencia} \\
\midrule
Complejidad Espacial             & $O(|V|^2)$           & $O(|V| + |E|)$            \\
Añadir una arista                & $O(1)$               & $O(1)$                \\
Eliminar una arista              & $O(1)$               & $O(\deg(v))$          \\
¿Existe arista $(u,v)$?          & $O(1)$               & $O(\deg(v))$          \\
Enumerar vecinos                 & $O(|V|)$             & $O(\deg(v))$          \\
Enumerar aristas                 & $O(|V|^2)$           & $O(|V| + |E|)$            \\
Obtener peso                     & $O(1)$               & $O(\deg(v))$          \\
Recorrido DFS/BFS                & $O(|V|^2)$           & $O(|V| + |E|)$            \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}\frametitle{Uso de Memoria}
\begin{itemize}
\item Matriz de adyacencia
    \begin{itemize}
    \item Siempre espacio $|V|^2$ (se puede reducir si no es dirigido)
    \item Buena para grafos densos donde $|E| \sim |V|^2$
    \end{itemize}
\item Lista de adyacencia
    \begin{itemize}
    \item Espacio proporcional a $|V| + |E|$
    \item Mejor para grafos dispersos donde $|E| \ll |V|^2$
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}\frametitle{Casos de Uso}
\centering
\begin{tabular}{@{} p{0.6\textwidth} l @{}}
\toprule
\textbf{Caso de Uso} & \textbf{Mejor Representación} \\
\midrule
El grafo es denso & Matriz de Adyacencia \\
El grafo es disperso & Lista de Adyacencia \\
Comprobar existencia de arista a menudo & Matriz de Adyacencia \\
Iterar sobre vecinos frecuentemente & Lista de Adyacencia \\
Minimizar uso de espacio & Lista de Adyacencia \\
Usar algoritmo de Dijkstra & Lista de Adyacencia \\
Usar algoritmo de Floyd-Warshall & Matriz de Adyacencia \\
\bottomrule
\end{tabular}
\end{frame}
