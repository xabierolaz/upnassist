\subsection{Trabajando con \texttt{list}}

\begin{frame}[fragile]\frametitle{Desempaquetado de una \texttt{list}}
\begin{block}{}
Podemos usar un \lstinline|*| antes del nombre de una variable para desempaquetar una lista en sus elementos individuales
\end{block}
\begin{lstlisting}
lst = ['uno', 'dos', 'tres', 'cuatro']

for elemento in lst:
    print(elemento, end=' ')
# Es equivalente a
print(*lst)
# Estamos llamando a print con todos los elementos de la lista como parámetros
print(lst[0], lst[1], lst[2], lst[3])
\end{lstlisting}
\begin{itemize}
\item Podemos trabajar con los parámetros \lstinline|end| y \lstinline|sep| para modificar cómo se muestra la lista
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Asignación de Variables Múltiples}
\begin{lstlisting}
a, b = 1, 2

def devuelve_dos():
    # Hacer algo
    return 'uno', 'dos'

primero, segundo = devuelve_dos()
\end{lstlisting}
\begin{itemize}
\item Python permite asignar más de una variable en una sola sentencia
\item Intenta hacer coincidir la estructura del lado izquierdo con el lado derecho
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Asignación de Variables con Desempaquetado}
\begin{itemize}
\item Podemos usar el mismo truco con el operador de desempaquetado
\item Solo una variable puede llevar el asterisco
\end{itemize}
\begin{lstlisting}
lst = ['uno', 'dos', 'tres', 'cuatro']

primero, segundo, *resto = lst

*inicio, ultimo = lst

primero, *medio, ultimo = lst
\end{lstlisting}

\end{frame}

\begin{frame}[fragile]\frametitle{Uso de \texttt{enumerate}}
\begin{itemize}
\item Al iterar por una lista podemos necesitar el índice o el elemento
\end{itemize}
\begin{lstlisting}
lst = ['uno', 'dos', 'tres', 'cuatro']

for elemento in lst:
    # Conocemos el elemento, pero no el índice

for i in range(len(lst)):
    # Muy verboso, para acceder al elemento usamos lst[i]
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Uso de \texttt{enumerate}}
\begin{itemize}
\item Al iterar por una lista podemos necesitar el índice o el elemento
\end{itemize}
\begin{lstlisting}
lst = ['uno', 'dos', 'tres', 'cuatro']

for cuenta, valor in enumerate(lst):
    # Tenemos el índice y el elemento al mismo tiempo
\end{lstlisting}
\end{frame}


\subsection{Lectura de Ficheros}

\begin{frame}\frametitle{¿Por qué usar ficheros?}
\begin{itemize}
\item Añadir persistencia a un programa
\item Procesar la salida de otro programa
\item Añadir opciones de configuración
\end{itemize}

\end{frame}

\begin{frame}[fragile]\frametitle{Interactuando con Ficheros}
\begin{itemize}
\item La mayoría de las operaciones de ficheros se hacen a través del módulo \lstinline|os|\footnote{\href{https://docs.python.org/3/library/os.html}{https://docs.python.org/3/library/os.html}}
\item La interacción más básica es comprobar si un fichero existe
\end{itemize}
\begin{lstlisting}[basicstyle=\small]
import os.path
nombre_fichero = "mi_fichero.txt"
if os.path.isfile(nombre_fichero):
    print("El fichero existe")
else
    print("El fichero no existe")
\end{lstlisting}
\begin{itemize}
\item \lstinline|os.path.exists(ruta_fichero)| también comprueba directorios
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Ayuda, no puedo acceder a mi fichero}
\begin{itemize}
\item Un fichero se localiza por su \textbf{ruta} (path)
    \begin{itemize}
    \item Ruta absoluta: 
        \begin{itemize}
        \item \verb*|/home/ugaitz/Desktop/mi_fichero.txt|
        \item \verb*|C:\\Users\\Ugaitz\\Desktop\\mi_fichero.txt|
        \end{itemize}
    \item Ruta relativa:
        \begin{itemize}
        \item Siempre relativa al directorio de trabajo actual (por defecto la carpeta donde ejecutas el archivo python)
        \item \verb*|mi_fichero.txt|
        \end{itemize}
    \end{itemize}
\item Intenta usar rutas relativas
\item Si no se encuentra un fichero, comprueba el directorio de trabajo actual del programa
\begin{lstlisting}[basicstyle=\small]
import os
cwd = os.getcwd()
\end{lstlisting}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Apertura de un Fichero}
\begin{lstlisting}[basicstyle=\small]
open(nombre_fichero, mode='r')
\end{lstlisting}
\small
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Modo} & \textbf{Descripción} \\
\midrule
\texttt{r}    & Abre el fichero de texto para lectura \\
\texttt{r+}   & Abre el fichero para lectura y escritura \\
\texttt{w}    & Abre el fichero para escritura. Vacía el fichero si ya existe. Crea uno nuevo si no existe. \\
\texttt{w+}   & Abre el fichero para lectura y escritura. Vacía el fichero si ya existe. Crea uno nuevo si no existe. \\
\texttt{a}    & Abre el fichero para escritura. Los datos escritos se insertarán al final del fichero. Crea uno nuevo si no existe. \\
\texttt{a+}   & Abre el fichero para lectura y escritura. Los datos escritos se insertarán al final del fichero. Crea uno nuevo si no existe. \\
\bottomrule
\end{tabularx}
\end{frame}


\begin{frame}[fragile]\frametitle{Apertura de un Fichero}
\begin{block}{}
\centering Se recomienda comprobar si el fichero existe antes de abrirlo para leer
\end{block}
\begin{lstlisting}[basicstyle=\small]
fichero = open("mi_fichero.txt", mode="r")
# Hacer cosas
fichero.close() # No olvides cerrar el fichero
\end{lstlisting}
\begin{lstlisting}[basicstyle=\small]
with open("mi_fichero.txt", mode="r") as fichero:
    # Hacer cosas
# Al salir del bloque with, el fichero se cierra automáticamente
\end{lstlisting}
\end{frame}


\begin{frame}[fragile]\frametitle{Lectura de Contenidos}
\begin{itemize}
\item \lstinline|file.readline() -> str|
    \begin{itemize}
    \item Lee una sola línea
    \end{itemize}
\item \lstinline|file.read() -> str|
    \begin{itemize}
        \item Lee el fichero completo y lo guarda en un string
    \end{itemize}
\item \lstinline|file.readlines() -> lst[str]|
    \begin{itemize}
        \item Lee el fichero completo y lo guarda en una lista de líneas
        \item Combinación de \lstinline|file.read()| y \lstinline|str.splitlines()|
    \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}[fragile]\frametitle{Iterar sobre un Fichero}
\begin{itemize}
\item Podemos iterar sobre cada línea de un fichero con una sentencia \lstinline|for|
\end{itemize}
\begin{lstlisting}[basicstyle=\small]
with open("mi_fichero.txt", mode="r") as fichero:
    for linea in fichero:
        print(linea)
        # es posible que tengamos que usar la función str.strip() para eliminar el salto de línea final
\end{lstlisting}
\end{frame}



\subsection{Uso de F-Strings}

\begin{frame}[fragile]\frametitle{Formateo de Salida}
\begin{lstlisting}[basicstyle=\small]
print('Hola', nombre, '! Bienvenido a', asignatura, '.')

print('Hola %s! Bienvenido a %s.' % (nombre, asignatura))

print('Hola {}! Bienvenido a {}.'.format(nombre, asignatura))

print(f'Hola {nombre}! Bienvenido a {asignatura}.')
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Uso de F-Strings}
\begin{itemize}
\item Creamos una F-String añadiendo f al principio de cualquier cadena
\item Dentro de las llaves podemos escribir lo que queramos
    \begin{itemize}
    \item Debe ser código python válido
    \end{itemize}
\end{itemize}
\begin{lstlisting}
print(f'Hola {nombre.upper()}! Bienvenido a {getAsignatura(nombre)}.')
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Formateo de F-Strings}
\begin{itemize}
\item Podemos pedir a python que formatee la salida de varias maneras
\item Sigue la misma \href{https://docs.python.org/3/library/string.html#format-specification-mini-language}{\color{blue}\underline{especificación}} que \lstinline|str.format()|
\item Usa dos puntos para separar el valor del formato
\begin{lstlisting}[basicstyle=\small]
numero = 123.123141241241
# El punto especifica el número de decimales
# La f al final es para tratarlo como float
print(f'Número con 3 decimales={numero:.3f}')

# Podemos especificar la dirección con <, >, ^
# Podemos reservar un espacio de 10 de ancho para el número
print(f'Más ancho y a la derecha={numero:>10.3f}')
\end{lstlisting}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Formateo de F-Strings, Uso Básico}
\begin{block}{Estructura de formato básica, todos los elementos son opcionales}
\centering\verb|[relleno][alineación][ancho][.decimales][tipo]|
\end{block}
\begin{description}
\item[relleno:] se usa si hay espacio extra reservado pero no usado, por defecto \lstinline|' '| 
\item[alineación:] se usa para especificar la alineación del texto
\item[ancho:] se usa para reservar un ancho específico para el texto, por defecto el requerido para mostrar la variable
\item[.decimales:] se usa para especificar cuántos decimales mostrar, el punto es obligatorio si lo usamos. Solo funciona con tipos numéricos
\item[tipo:] se usa para forzar el tipo de la variable a mostrar
\end{description}
\end{frame}

\subsection{Pistas de Tipo (Type Hinting)}

\begin{frame}\frametitle{Tipado en Python}
\begin{itemize}
\item Python no pide al programador especificar tipos de variables
    \begin{itemize}
    \item Lo hace en tiempo de ejecución
    \end{itemize}
\item Podemos especificar el tipo de las variables, parámetros de función o retornos
    \begin{itemize}
    \item No es obligatorio
    \item Útil como documentación
    \item Python no lo tiene en cuenta al ejecutar el código
    \item Introducido en la versión 3.5, mejorado en la 3.9
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Tipado Simple}
\begin{block}{}
\centering Podemos añadir un tipo a cualquier variable añadiendo dos puntos tras el nombre
\end{block}
\begin{lstlisting}
a: int = 5
b: float = 2.3
c: list = []

def get(lst: list, i: int, j: int) -> int:
    # hacer algo
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Complicando los Tipos}
\begin{itemize}
\item Desde la versión 3.9 de Python podemos especificar el contenido de las colecciones
\item Para leer más ve \href{https://mypy.readthedocs.io/en/stable/cheat_sheet_py3.html}{\color{blue}\underline{aquí}}
\begin{lstlisting}
# m es una lista de listas que contienen enteros
m: list[list[int]] = ...

# d es un diccionario con strings como claves y floats como valores
d: dict[str, float] = ...

# podemos hacerlos aún más complejos
a: dict[str, list[list[int]]] = ...
\end{lstlisting}
\end{itemize}
\end{frame}

\subsection{Definiendo una Función \texttt{main}}

\begin{frame}\frametitle{La Variable \texttt{\_\_name\_\_}}
\begin{itemize}
\item Cada vez que ejecutamos un archivo en Python se crean varias variables en segundo plano
\item Una de ellas es la variable \lstinline|__name__|. Esta variable tiene una de estas dos informaciones:
    \begin{description}
    \item[nombre\_fichero:] Si no estamos ejecutando directamente el archivo (lo estamos importando), su variable \lstinline|__name__| se define como el nombre del archivo
    \item[\texttt{\_\_main\_\_}:] Si estamos ejecutando directamente el archivo, entonces su \lstinline|__name__| contendrá el string \lstinline|'__main__'|
    \end{description} 
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Definiendo el Script Ejecutado}
\begin{itemize}
\item Usando la variable \lstinline|__name__| podemos diferenciar entre lanzar el script directamente o importarlo
\item Podemos tener trozos de código que se ejecuten solo si el script no está siendo importado
    \begin{itemize}
    \item Por ejemplo para propósitos de prueba
    \end{itemize}
\end{itemize}
\begin{lstlisting}
def main():
    # Esta función es el punto de entrada al programa

# Usaremos este trozo de código al final del archivo
if __name__ == "__main__":
    main()
\end{lstlisting}
\end{frame}


\section{Creando y Usando una Clase}

\begin{frame}[fragile]\frametitle{¿Qué es una Clase?}
\begin{itemize}
\item La mayor parte de los internos de Python están definidos usando clases
    \begin{itemize}
    \item \lstinline|list, dict, set, int, float| y más
    \end{itemize}
\item Una clase es una plantilla que define un comportamiento para un elemento con un equivalente en la ``vida real''
\item Cuando programamos estamos creando instancias de clases
    \begin{itemize}
    \item Estamos trabajando con \textbf{objetos}
    \end{itemize}
\end{itemize}
\begin{lstlisting}
# Creamos un objeto de la clase lista
lst = list()
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Partes de una Clase}
\begin{itemize}
\item Una clase se divide en dos componentes principales
\begin{description}
\item[Atributos:] Variables que están dentro de la clase, el acceso directo a los atributos de una clase suele estar desaconsejado
\item[Métodos:] Funciones que se definen dentro de la clase, pueden usar los atributos en su ejecución
\end{description}
\end{itemize}
\begin{lstlisting}
lst = list()
# Llamamos al método append() de la clase lista
lst.append(5)

# Sabemos que lst contiene el elemento en algún atributo, pero no sabemos cómo
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Creando una Clase}
\begin{itemize}
\item Creamos una clase usando la palabra clave \lstinline|class|
\item El método \lstinline|__init__| es una función especial que se llama siempre que creamos un objeto de esa clase
    \begin{itemize}
    \item Siempre que hacemos \lstinline|var = MiClase()|
    \end{itemize}
\item El método \lstinline|__init__| también se usa para crear los atributos de una clase
    \begin{itemize}
    \item \lstinline|self| se refiere al objeto que se está usando
    \item Los guiones bajos en los nombres de los atributos dicen a cualquier otro programador que no los acceda directamente
    \end{itemize}
\end{itemize}
\begin{lstlisting}
class MiClase:
    def __init__(self, param1, param2):
        self.__attr1 = param1
        self.__attr2 = param2
\end{lstlisting}
\end{frame}


\begin{frame}[fragile]\frametitle{¿Por qué el \ \_\_?}
\begin{itemize}
\item Python no ofrece una forma de controlar quién puede acceder a los atributos a diferencia de la mayoría de los otros lenguajes de programación orientados a objetos
    \begin{itemize}
    \item \texttt{público}
        \begin{itemize}
        \item Cualquiera puede acceder y modificar el atributo
        \end{itemize}
    \item \texttt{protegido}
        \begin{itemize}
        \item El nombre del atributo empieza por un guion bajo
        \item El atributo se usa internamente por la clase, pero podemos si queremos
        \end{itemize}
    \item \texttt{privado}
        \begin{itemize}
        \item El nombre del atributo empieza por un doble guion bajo
        \item El atributo se usa internamente por la clase, se realizan algunas modificaciones en su nombre
        \item Usaremos esto para nuestros atributos
        \end{itemize}
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Añadiendo un Método}
\begin{itemize}
\item Todos los métodos tendrán \lstinline|self| como su primer parámetro
\item Pueden tener tantos parámetros como sean necesarios
\item Pueden llamar a cualquier otra función o método
\end{itemize}
\begin{lstlisting}
class MiClase:
    def __init__(self, param1, param2):
        self.__attr1 = param1
        self.__attr2 = param2
    def miMetodo1(self, param):
        # Cálculos complicados
        self.__attr1 = param
    def miMetodo2(self, param):
        self.miMetodo1(param)
        self.__attr2 = param
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Usando un Método}
\begin{itemize}
\item Siempre que llamamos a un método, Python maneja automáticamente la definición del parámetro self
\item Si lo hacemos a mano no estamos usando la clase correctamente
\end{itemize}
\begin{lstlisting}
lst = list()
# Llamar al método
lst.append(5)
# Llamada errónea al método
list.append(lst, 5)

var = MiClase(5, 6)
var.miMetodo2(8)
\end{lstlisting}
\end{frame}
