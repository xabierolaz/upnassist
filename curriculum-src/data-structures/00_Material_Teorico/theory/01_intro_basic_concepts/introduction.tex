\section{Reviewing and Expanding Python Code}

\subsection{Working with \texttt{list}}

\begin{frame}[fragile]\frametitle{Unpacking a \texttt{list}}
\begin{block}{}
We can use an \lstinline|*| before a variable name to unpack a list into its individual elements
\end{block}
\begin{lstlisting}
lst = ['one', 'two', 'three', 'four']

for element in lst:
    print(element, end=' ')
# Is equivalent to
print(*lst)
# We are calling print with all elements on the list as parameters
print(lst[0], lst[1], lst[2], lst[3])
\end{lstlisting}
\begin{itemize}
\item We can work with the \lstinline|end| and \lstinline|sep| parameters to modify how the list is shown
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Multiple Variable Assignment}
\begin{lstlisting}
a, b = 1, 2

def returns_two():
    # Do something
    return 'one', 'two'

first, second = returns_two()
\end{lstlisting}
\begin{itemize}
\item Python allows assigning more than one variable in a single statement
\item It tries to match the structure on the left-hand side to the right-hand side
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Variable Assignment with Unpacking}
\begin{itemize}
\item We can use the same trick with the unpacking operator
\item Only one variable can be starred
\end{itemize}
\begin{lstlisting}
lst = ['one', 'two', 'three', 'four']

first, second, *rest = lst

*start, last = lst

first, *middle, last = lst
\end{lstlisting}

\end{frame}

\begin{frame}[fragile]\frametitle{Using \texttt{enumerate}}
\begin{itemize}
\item When iterating through a list we might need the index or the element
\end{itemize}
\begin{lstlisting}
lst = ['one', 'two', 'three', 'four']

for element in lst:
    # We know the element, but not the index

for i in range(len(lst)):
    # Very verbose, to acces the element we use lst[i]
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Using \texttt{enumerate}}
\begin{itemize}
\item When iterating through a list we might need the index or the element
\end{itemize}
\begin{lstlisting}
lst = ['one', 'two', 'three', 'four']

for count, value in enumerate(lst):
    # We have the index and the element at the same time
\end{lstlisting}
\end{frame}


\subsection{Reading Files}

\begin{frame}\frametitle{Why Use Files}
\begin{itemize}
\item Adding persistence to a program
\item Processing output from another program
\item Adding configuration options
\end{itemize}

\end{frame}

\begin{frame}[fragile]\frametitle{Interacting with Files}
\begin{itemize}
\item Most file operations are done through the \lstinline|os|\footnote{\href{https://docs.python.org/3/library/os.html}{https://docs.python.org/3/library/os.html}} module
\item The most basic interaction is to check if a file exists
\end{itemize}
\begin{lstlisting}[basicstyle=\small]
import os.path
filename = "my_file.txt"
if os.path.isfile(filename):
    print("File exists")
else
    print("File does not exist")
\end{lstlisting}
\begin{itemize}
\item \lstinline|os.path.exists(file_path)| also checks for directories
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Help I Cannot Access my File}
\begin{itemize}
\item A file is located by its \textbf{path}
    \begin{itemize}
    \item Absolute path: 
        \begin{itemize}
        \item \verb*|/home/ugaitz/Desktop/my_file.txt|
        \item \verb*|C:\Users\Ugaitz\Desktop\my_file.txt|
        \end{itemize}
    \item Relative path:
        \begin{itemize}
        \item Always relative to the current working directory (by default the folder where you execute the python file)
        \item \verb*|my_file.txt|
        \end{itemize}
    \end{itemize}
\item Try to use relative paths
\item If a file is not found check the current working directory for the program
\begin{lstlisting}[basicstyle=\small]
import os
cwd = os.getcwd()
\end{lstlisting}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Opening a File}
\begin{lstlisting}[basicstyle=\small]
open(filename, mode='r')
\end{lstlisting}
\small
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Mode} & \textbf{Description} \\ 
\midrule
\texttt{r}    & Open text file for reading \\
\texttt{r+}   & Open the file for reading and writing \\
\texttt{w}    & Open the file for writing. Truncates the file if it already exists. Creates a new file if it does not exist. \\
\texttt{w+}   & Open the file for reading and writing. Truncates the file if it already exists. Creates a new file if it does not exist. \\
\texttt{a}    & Open the file for writing. The data being written will be inserted at the end of the file. Creates a new file if it does not exist. \\
\texttt{a+}   & Open the file for reading and writing. The data being written will be inserted at the end of the file. Creates a new file if it does not exist. \\
\bottomrule
\end{tabularx}
\end{frame}


\begin{frame}[fragile]\frametitle{Opening a File}
\begin{block}{}
\centering It is recommended to check if the file exists before opening to read
\end{block}
\begin{lstlisting}[basicstyle=\small]
file = open("my_file.txt", mode="r")
# Do stuff
file.close() # Do not forget to close the file
\end{lstlisting}
\begin{lstlisting}[basicstyle=\small]
with open("my_file.txt", mode="r") as  file:
    # Do stuff
# When we exit the with statement the file is closed automaticaly
\end{lstlisting}
\end{frame}


\begin{frame}[fragile]\frametitle{Read the Contents}
\begin{itemize}
\item \lstinline|file.readline() -> str|
    \begin{itemize}
    \item Read a single line
    \end{itemize}
\item \lstinline|file.read() -> str|
    \begin{itemize}
        \item Read the whole file and store it in a string
    \end{itemize}
\item \lstinline|file.readlines() -> lst[str]|
    \begin{itemize}
        \item Read the whole file and store it in a list of lines
        \item Combination of \lstinline|file.read()| and \lstinline|str.splitlines()|
    \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}[fragile]\frametitle{Iterate Over a File}
\begin{itemize}
\item We can iterate over each line of a file with a \lstinline|for| statement
\end{itemize}
\begin{lstlisting}[basicstyle=\small]
with open("my_file.txt", mode="r") as  file:
    for line in file:
        print(line)
        # we might have to use the str.strip() function to remove any trailing newline
\end{lstlisting}
\end{frame}



\subsection{Using F-Strings}

\begin{frame}[fragile]\frametitle{Formatting Output}
\begin{lstlisting}[basicstyle=\small]
print('Hello', name, '! Welcome to', subject, '.')

print('Hello %s! Welcome to %s.' % (name, subject))

print('Hello {}! Welcome to {}.'.format(name, subject))

print(f'Hello {name}! Welcome to {subject}.')
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Using F-Strings}
\begin{itemize}
\item We create an F-String by adding f to the beginning of any string
\item Inside the curly brackets we can write anything we want
    \begin{itemize}
    \item It has to be valid python code
    \end{itemize}
\end{itemize}
\begin{lstlisting}
print(f'Hello {name.upper()}! Welcome to    {getSubject(name)}.')
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Formatting F-Strings}
\begin{itemize}
\item We can ask python to format the output in several ways
\item Follows the same \href{https://docs.python.org/3/library/string.html#format-specification-mini-language}{\color{blue}\underline{specification}} as \lstinline|str.format()|
\item Use a colon to separate the value from the format
\begin{lstlisting}[basicstyle=\small]
number = 123.123141241241
# The point specifies number of decimals
# The f at the end is to treat it as float
print(f'Number with 3 decimals={number:.3f}')

# We can specify direction with <, >, ^
# We can reserve 10 wide space for the number
print(f'Wider and right={number:>10.3f}')
\end{lstlisting}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Formatting F-Strings, Basic Usage}
\begin{block}{Basic format structure, all elements are optional}
\centering\verb|[fill][align][width][.decimals][type]|
\end{block}
\begin{description}
\item[fill:] used if there is extra space reserved but not used, by default \lstinline|' '|
\item[align:] used to specify the alignment of the text
\item[width:] used to reserve a specific with for the text, by default the required to show the variable
\item[.decimals:] used to specify how many decimals to show, the dot is mandatory if we use it. Only works with numeric types
\item[type:] used to coerce the type of the variable to show
\end{description}
\end{frame}

\subsection{Type Hinting}

\begin{frame}\frametitle{Typing in Python}
\begin{itemize}
\item Python does not ask the programmer to specify variable types
    \begin{itemize}
    \item It does so at runtime
    \end{itemize}
\item We can specify the type of variables, function parameters, or returns
    \begin{itemize}
    \item It is not mandatory
    \item Useful as documentation
    \item Python does not take it into account when executing the code
    \item Introduced with version 3.5, improved on 3.9
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Simple Typing}
\begin{block}{}
\centering We can add a type to any variable by adding a colon after the name
\end{block}
\begin{lstlisting}
a: int = 5
b: float = 2.3
c: list = []

def get(lst: list, i: int, j: int) -> int:
    # do something
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Complicating the Types}
\begin{itemize}
\item Since Python version 3.9 we can specify the contents of the collections
\item To read more go \href{https://mypy.readthedocs.io/en/stable/cheat_sheet_py3.html}{\color{blue}\underline{here}}
\begin{lstlisting}
# m is a list of lists that contain integers
m: list[list[int]] = ...

# d is a dictionary with strings as keys and floats as values
d: dict[str, float] = ...

# we can make them even more complex
a: dict[str, list[list[int]]] = ...
\end{lstlisting}
\end{itemize}
\end{frame}

\subsection{Defining a \texttt{main} Function}

\begin{frame}\frametitle{The \texttt{\_\_name\_\_} Variable}
\begin{itemize}
\item Each time we execute a file in Python there are several variables that are created in the background
\item One of them is the \lstinline|__name__| variable. This variable has one of two pieces of information:
    \begin{description}
    \item[filename:] If we are not directly executing the file (we are importing it), its \lstinline|__name__| variable is defined as the filename
    \item[\texttt{\_\_main\_\_}:] If we are directly executing the file then its \lstinline|__name__| will contain the string \lstinline|'__main__'|
    \end{description} 
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Defining the Executed Script}
\begin{itemize}
\item Using the \lstinline|__name__| variable we can differentiate between launching the script directly or importing it
\item We can have pieces of code that are executed only if the script is not being imported
    \begin{itemize}
    \item For example for testing purposes
    \end{itemize}
\end{itemize}
\begin{lstlisting}
def main():
    # This function is the entry point to the program

# We will use this piece of code at the end of the file
if __name__ == "__main__":
    main()
\end{lstlisting}
\end{frame}


\section{Creating and Using a Class}

\begin{frame}[fragile]\frametitle{What is a Class?}
\begin{itemize}
\item Most of Python's internals are defined using classes
    \begin{itemize}
    \item \lstinline|list, dict, set, int, float| and more
    \end{itemize}
\item A class is a template that defines a behaviour for an element with a ``real life'' equivalent
\item When we program we are creating instances of classes
    \begin{itemize}
    \item We are working with \textbf{objects}
    \end{itemize}
\end{itemize}
\begin{lstlisting}
# We create an objet of class list
lst = list()
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Parts of a Class}
\begin{itemize}
\item A class is divided into two main components
\begin{description}
\item[Attributes:] Variables that are inside the class, direct access to the attributes of a class is often disallowed
\item[Methods:] Functions that are defined inside the class, they might use the attributes in their execution
\end{description}
\end{itemize}
\begin{lstlisting}
lst = list()
# We call the append() method of the list class
lst.append(5)

# We know that lst contains the element in some atribute, but we do not know how
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Creating a Class}
\begin{itemize}
\item We create a class by using the \lstinline|class| keyword
\item The \lstinline|__init__| method is a special function that is called whenever we create an object of that class
    \begin{itemize}
    \item Whenever we do \lstinline|var = MyClass()|
    \end{itemize}
\item The \lstinline|__init__| method is also used to create the attributes for a class
    \begin{itemize}
    \item \lstinline|self| refers to the object that is being used
    \item The underscores on the attribute names tell any other programmer no to access them directly
    \end{itemize}
\end{itemize}
\begin{lstlisting}
class MyClass:
    def __init__(self, param1, param2):
        self.__attr1 = param1
        self.__attr2 = param2
\end{lstlisting}
\end{frame}


\begin{frame}[fragile]\frametitle{Why the \_\_?}
\begin{itemize}
\item Python does not offer a way to control who can access attributes unlike most other object-oriented programming languages
    \begin{itemize}
    \item \texttt{public}
        \begin{itemize}
        \item Anyone can access and modify the attribute
        \end{itemize}
    \item \texttt{protected}
        \begin{itemize}
        \item The attribute's name starts with an underscore
        \item The attribute is used internally by the class, but we can if we want
        \end{itemize}
    \item \texttt{private}
        \begin{itemize}
        \item The attribute's name starts with a double underscore
        \item The attribute is used internally by the class, some modifications are done to its name
        \item We will use this for our attributes
        \end{itemize}
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Adding a Method}
\begin{itemize}
\item All methods will have \lstinline|self| as their first parameter
\item They can have as many parameters as necessary
\item They can call any other function or method
\end{itemize}
\begin{lstlisting}
class MyClass:
    def __init__(self, param1, param2):
        self.__attr1 = param1
        self.__attr2 = param2
    def myMethod1(self, param):
        # Complicated calculations
        self.__attr1 = param
    def myMethod2(self, param):
        self.myMethod1(param)
        self.__attr2 = param
\end{lstlisting}\textbf{}
\end{frame}

\begin{frame}[fragile]\frametitle{Using a Method}
\begin{itemize}
\item Whenever we call a method Python automatically handles defining the self parameter
\item If we do it by hand we are not using the class correctly
\end{itemize}
\begin{lstlisting}
lst = list()
# Call the method
lst.append(5)
# Erroneous call to the method
list.append(lst, 5)

var = MyClass(5, 6)
var.myMethod2(8)
\end{lstlisting}
\end{frame}
