\begin{frame}\frametitle{Características}
\begin{itemize}
	\item Cada rama de un árbol crea un \textbf{subárbol}
	    \begin{itemize}
	    	\item Podemos tratarlo como un árbol más pequeño
	    	\item La estructura de árbol se define recursivamente
	    \end{itemize}
	\item Podemos definir dos características para cualquier árbol:
	\begin{description}
	\item[Altura:] Número máximo de niveles que tiene el árbol
	\item[Peso:] Número de nodos que tiene el árbol
	\end{description}
\end{itemize}
\end{frame}

\section{Árboles Binarios}

\subsection{¿Qué es un Árbol Binario?}

\begin{frame}\frametitle{Limitando un Árbol}
\begin{itemize}
	\item Como con cualquier estructura, podemos añadir especificaciones sobre cómo usamos un árbol
	\item Una restricción común es limitar cada nodo a \textbf{dos hijos}
	    \begin{itemize}
	    	\item Debido a la naturaleza recursiva de los árboles, esto significa que cada nodo tiene dos subárboles
	    	\item Podemos nombrar a cada hijo como \textbf{izquierdo} o \textbf{derecho}
	    \end{itemize}
	\end{itemize}
	\pause
	\begin{block}{}
	\centering\Large Árbol Binario
	\end{block}
\end{frame}

	\item[Pre-orden:] Primero el valor del nodo, luego el subárbol izquierdo y finalmente el derecho

	\item[Post-orden:] Primero el subárbol izquierdo, luego el derecho y finalmente el valor del nodo
\end{description}
\end{frame}

\begin{frame}\frametitle{Limitando la Inserción}
\begin{itemize}
	\item Podemos limitar más cómo manejamos la inserción eligiendo una posición basada en el valor a insertar
	\item Un Árbol Binario de Búsqueda añade solo una pequeña restricción
	    \begin{itemize}
	    	\item El subárbol izquierdo contiene valores que son menores
	    	\item El subárbol derecho contiene valores que son mayores
	    \end{itemize}
	\item No podremos especificar dónde insertar
	    \begin{itemize}
	    	\item El árbol elegirá una posición
	    	\item Insertará recursivamente a izquierda o derecha
	    \end{itemize}
	\item El principal beneficio es que obtenemos un acceso rápido a los datos
	    \begin{itemize}
	    	\item Solo tenemos que mirar una fracción del árbol
	    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}\frametitle{Un Pequeño Cambio Crea Grandes Complicaciones}
\begin{itemize}
	\item ¿Cómo podemos borrar un nodo?
\begin{description}
	\item[El nodo es una hoja:] Simplemente lo borramos
	\item[El nodo tiene 1 hijo:] El hijo se convierte en el nodo
	\item[El nodo tiene 2 hijos:] Encontrar el sucesor in-orden (el predecesor también sirve), intercambiar posiciones y luego borrar el nodo.
\end{description}
\end{itemize}
\end{frame}
