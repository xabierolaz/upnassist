\section{Tipos de Datos Abstractos}

\begin{frame}\frametitle{¿Qué es un \textit{TAD}?}
\begin{itemize}
\item Un \textit{Tipo de Dato Abstracto} (TAD) es una forma de modelar la interacción que un usuario tiene con los datos
\item La estructura se define solo conceptualmente
    \begin{itemize}
    \item Valores posibles
    \item Operaciones sobre el tipo de dato y su comportamiento
    \end{itemize}
\item Un TAD describe la forma lógica de los datos, una \textbf{estructura de datos} es su implementación
\end{itemize}
\end{frame}

\section{\textit{TADs} implementados en Python}

\begin{frame}\frametitle{Algunos TADs ya están implementados en Python}
\begin{itemize}
\item La mayoría de los lenguajes de programación ya implementan algunos TADs
\item Python tiene varios en sus librerías núcleo:
    \begin{itemize}
    \item Listas
    \item Mapas (Diccionarios)
    \item Conjuntos (Sets)
    \end{itemize}
\item Hay más en librerías externas
    \begin{itemize}
    \item Colas/Deques
    \item Árboles
    \item Grafos
    \end{itemize}
\end{itemize}
\end{frame}

\subsection{Listas}

\begin{frame}\frametitle{El TAD Lista}
\begin{itemize}
\item La \texttt{Lista} es uno de los TAD más comunes
\item Almacena puntos de datos en un orden dado
    \begin{itemize}
    \item La longitud de la lista suele ser dinámica
    \end{itemize}
\item Métodos básicos de \texttt{Lista}
    \begin{itemize}
    \item Crear una lista vacía
    \item Comprobar si la lista está vacía
    \item Añadir un elemento a la lista
    \item Eliminar un elemento de la lista
    \item Acceder a un elemento con un índice
    \end{itemize}
\end{itemize}
\vspace{0.4cm}
\begin{center}

  \node[list,on chain] (B) {99};

\end{frame}

\subsection{Mapas}

\begin{frame}\frametitle{El TAD Mapa}
\begin{itemize}
\item Un \texttt{Mapa} es una colección de pares $<clave, valor>$
\item Puede implementarse usando tablas hash, árboles de búsqueda u otras estructuras
\item Métodos básicos de \texttt{Mapa}:
    \begin{itemize}
    \item Insertar un nuevo par $<clave, valor>$
    \item Eliminar un par $<clave, valor>$
    \item Obtener un valor a partir de una clave dada
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}\frametitle{Mapas en Python}
\Wider{\centering\scriptsize
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Método} & \textbf{Uso} & \textbf{Explicación} \\
\midrule
 & \texttt{a\_dict[key] = value} & Establece un valor para una clave \\
 & \texttt{a\_dict[key]} & Devuelve el valor asociado a la clave \\
\texttt{keys} & \texttt{a\_dict.keys()} & Devuelve las claves en un objeto \texttt{dict\_keys} \\
\texttt{values} & \texttt{a\_dict.values()} & Devuelve los valores en un objeto \texttt{dict\_values} \\
\texttt{items} & \texttt{a\_dict.items()} & Devuelve los pares clave-valor en \texttt{dict\_items} \\
\texttt{get} & \texttt{a\_dict.get(key)} & Devuelve el valor o \texttt{None} si no existe \\
\texttt{get} & \texttt{a\_dict.get(key, alt)} & Devuelve el valor o \texttt{alt} si no existe \\
\bottomrule
\end{tabular}
\normalsize

\section{Pila (Stack)}

\begin{frame}\frametitle{El TAD Pila}
\begin{itemize}
\item Last in First out (LIFO - Último en entrar, primero en salir)
\end{itemize}

\begin{frame}\frametitle{Operando en una Pila}
\begin{itemize}
\item La Pila tendrá al menos dos operaciones
    \begin{description}
    \item[push:] Añadir un nuevo elemento en la cima de la Pila
    \item[pop:] Eliminar y devolver el elemento de la cima de la Pila
    \end{description}
\item Puede haber más operaciones opcionales
    \begin{description}
    \item[peek:] Devolver el elemento de la cima sin eliminarlo
    \item[is\_empty:] Devuelve \lstinline|True| si está vacía, \lstinline|False| en caso contrario
    \item[size:] Devuelve el número de elementos en la Pila
    \end{description}
\end{itemize}
\end{frame}

\section{Cola (Queue)}

\begin{frame}\frametitle{El TAD Cola}
\begin{itemize}
\item First in First out (FIFO - Primero en entrar, primero en salir)
\end{itemize}
\vspace{0.4cm}

\begin{frame}\frametitle{Operando en una Cola}
\begin{itemize}
\item La Cola tendrá al menos dos operaciones
    \begin{description}
    \item[enqueue:] Añadir un nuevo elemento al final de la Cola
    \item[dequeue:] Eliminar y devolver un elemento del frente de la Cola
    \end{description}
\item Al igual que con las Pilas, una Cola puede tener más operaciones
    \begin{description}
    \item[peek:] Devolver el elemento frontal sin eliminarlo
    \item[is\_empty:] Devuelve \lstinline|True| si está vacía, \lstinline|False| en caso contrario
    \item[size:] Devuelve el número de elementos en la Cola
    \end{description}
\end{itemize}
\end{frame}
