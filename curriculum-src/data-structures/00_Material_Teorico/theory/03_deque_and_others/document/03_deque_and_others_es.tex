\section{Cola de Doble Extremo (Deque)}

\begin{frame}\frametitle{¿Qué es una Cola Doble?}
\begin{itemize}
\item Una cola de doble extremo, también conocida como Deque, es una generalización de una cola
    \begin{itemize}
    \item Los elementos pueden añadirse o eliminarse por el frente o por el final
    \item Se pronuncia como \textit{deck}
    \item En algunos casos puede aparecer como dequeue
    \end{itemize}
\item A veces se añaden restricciones sobre cómo se insertan o eliminan los ítems
    \begin{itemize}
    \item ¿Qué pasaría si prohibimos añadir o eliminar elementos solo por un lado?
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}\frametitle{Operaciones en un Deque}

\section{Cola de Prioridad}

\begin{frame}\frametitle{Cola de Prioridad}
\begin{itemize}
\item A veces un elemento en una cola tiene mayor prioridad que los demás
    \begin{itemize}
    \item Esto significa que los elementos deben ser desencolados antes que otros
    \item Colas en un parque de atracciones
    \item Embarque en un avión
    \end{itemize}
\item Cada vez que se encola un ítem \textbf{le damos una prioridad}
\item Para desencolar un ítem desencolamos el elemento más antiguo insertado con la prioridad más baja
\end{itemize}
\end{frame}

\section{Búfer Circular}

\begin{frame}\frametitle{Búfer Circular}
\begin{itemize}
\item Podemos limitar el tamaño de una cola
    \begin{itemize}
    \item A medida que se insertan nuevos elementos, se eliminan los antiguos
    \end{itemize}
\item En algunos casos solo nos interesan los datos más recientes
    \begin{itemize}
    \item Streaming de vídeo/audio
    \item Buffering de flujos de datos, de ahí el nombre
    \end{itemize}
\item Sigue FIFO
\item No hay principio ni fin en el búfer, solo para los datos que almacena
\end{itemize}
\end{frame}
