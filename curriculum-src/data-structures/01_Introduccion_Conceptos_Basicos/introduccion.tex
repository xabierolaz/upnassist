\subsection{Trabajando con \texttt{list}}

\begin{frame}[fragile]\frametitle{Controlando el final de línea (\texttt{end})}
\begin{lstlisting}[basicstyle=\scriptsize]
# 1. Comportamiento por defecto (Baja de linea):
lst = ['uno', 'dos']
for e in lst:
    print(e)
\end{lstlisting}
\terminal{uno\\dos}

\begin{lstlisting}[basicstyle=\scriptsize]
# 2. Usando end=' ' (Se queda en la misma linea):
for e in lst:
    print(e, end=' ')
\end{lstlisting}
\terminal{uno dos}
\end{frame}

\begin{frame}[fragile]\frametitle{Desempaquetado y Separadores (\texttt{sep})}
\begin{lstlisting}
# Desempaquetado con separador personalizado
lst = ['uno', 'dos', 'tres']
print(*lst, sep=' - ')
\end{lstlisting}
\terminal{uno - dos - tres}

\begin{lstlisting}
# Comparar con print normal (muestra el objeto lista)
print(lst)
\end{lstlisting}
\terminal{['uno', 'dos', 'tres']}
\end{frame}

\begin{frame}[fragile]\frametitle{Asignación Múltiple de Variables}
\begin{block}{El concepto}
Podemos asignar varios valores a varias variables en una sola línea.
\end{block}

\begin{columns}
\begin{column}{0.48\textwidth}
\textbf{Forma Python (Rápida)}
\begin{lstlisting}[basicstyle=\tiny]
nombre, edad = "Ana", 25
\end{lstlisting}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Forma Clásica (Lenta)}
\begin{lstlisting}
nombre = "Ana"
edad = 25
\end{lstlisting}
\end{column}
\end{columns}

\vspace{0.5cm}
\begin{itemize}
\item Python empareja por orden: el 1º valor va a la 1ª variable, el 2º a la 2ª.
\item Los tipos de dato pueden ser diferentes (texto y número).
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Asignación de Variables con Desempaquetado}
\lstinline|colores = ['azul', 'verde', 'rojo', 'negro']|
\begin{lstlisting}[basicstyle=\scriptsize]
# 1. El resto al final:
azul, verde, *resto = colores
print(otros)
\end{lstlisting}
\terminal{['rojo', 'negro']}
\begin{lstlisting}[basicstyle=\scriptsize]
# 2. El resto al principio:
*resto, negro = colores
print(otros)
\end{lstlisting}
\terminal{['azul', 'verde', 'rojo']}
\begin{lstlisting}[basicstyle=\scriptsize]
# 3. El resto en medio (Sandwich):
azul, *resto, negro = colores
print(otros)
\end{lstlisting}
\terminal{['verde', 'rojo']}
\end{frame}

\begin{frame}[fragile]\frametitle{El Problema: ¿En qué posición está?}
\begin{lstlisting}[basicstyle=\scriptsize]
# Intento 1: Bucle for normal
ranking = ['Ana', 'Beto']
for persona in ranking:
    print(persona)
\end{lstlisting}
\terminal{Ana\\Beto}
\begin{lstlisting}[basicstyle=\scriptsize]
# Intento 2: Usando indices manuales
for i in range(len(ranking)):
    # Acceso manual con [i] y correccion de indice (+1)
    print(i + 1, ":", ranking[i])
\end{lstlisting}
\terminal{1 : Ana\\2 : Beto}
\end{frame}

\begin{frame}[fragile]\frametitle{La Solución: enumerate}
\begin{itemize}
    \item \lstinline|enumerate| pega un número a cada elemento, creando \textbf{parejas} (tuplas).
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
#           0       1       2
ranking = ['Ana', 'Beto', 'Carla']

# 1. Veamos que genera realmente enumerate:
# 1. El indice (contador automatico)
# 2. El dato (nombre)

for i, nombre in enumerate(ranking):
    print(i + 1, ":", nombre)
\end{lstlisting}
\terminal{1 : Ana\\2 : Beto\\3 : Carla}
\end{frame}

\subsection{Leyendo Archivos}

\begin{frame}\frametitle{¿Por Qué Usar Archivos?}
\begin{itemize}
\item Añadir persistencia a un programa
\item Procesar la salida de otro programa
\item Añadir opciones de configuración
\end{itemize}

\end{frame}

\begin{frame}[fragile]\frametitle{Interactuando con Archivos}
El módulo \lstinline|os| es nuestro puente con el sistema operativo.
\begin{lstlisting}[basicstyle=\scriptsize]
import os

# 1. ¿Donde estoy? (Get Current Working Directory)
print(os.getcwd())

# 2. ¿Existe algo ahi? (Archivo O Carpeta)
ruta = "datos.txt"
if os.path.exists(ruta):
    print("Algo existe...")

# 3. ¿Es un archivo valido? (RECOMENDADO)
if os.path.isfile(ruta):
    print("¡Es un archivo! Puedo leerlo.")
else:
    print("No es un archivo (o no existe).")
\end{lstlisting}
\begin{itemize}\scriptsize
    \item \textbf{Nota:} \lstinline|exists()| devuelve \lstinline|True| también para carpetas (que \lstinline|open| no puede leer). Por seguridad, usa \textbf{isfile}.
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Ayuda, No Puedo Acceder a mi Archivo}
\begin{itemize}
\item \textbf{Rutas y Sistemas Operativos:}
    \begin{itemize}
    \item \textbf{Windows:} Usa \lstinline|\| (ej: \verb*|C:\datos\notas.txt|)
    \item \textbf{Linux/Mac:} Usa \lstinline|/| (ej: \verb*|/home/user/notas.txt|)
    \end{itemize}
\item \textbf{¡Cuidado!} En Python, \lstinline|\| puede dar errores (ej: \lstinline|\n| se lee como salto de línea).
\item \textbf{Truco Maestro:} Usa SIEMPRE la barra \lstinline|/|. Python la entiende en Windows y es más seguro.
\item \textbf{Rutas Relativas:} (Recomendado)
    \begin{itemize}
    \item \verb*|mi_archivo.txt| (En la misma carpeta)
    \item \verb*|subcarpeta/archivo.txt| (Dentro de una carpeta)
    \end{itemize}
\end{itemize}
\begin{lstlisting}[basicstyle=\tiny]
import os
print(os.getcwd()) # Si no carga, comprueba SIEMPRE donde estas
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Abriendo un Archivo}
\begin{lstlisting}[basicstyle=\small]
open(nombre_fichero, mode='r')
\end{lstlisting}
\small
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Modo} & \textbf{Descripción} \\ 
\midrule
\texttt{r}    & Abrir archivo de texto para lectura \\
\texttt{r+}   & Abrir el archivo para lectura y escritura \\
\texttt{w}    & Abrir archivo para escritura. \textbf{Sobreescribe} (borra lo anterior). Crea archivo si no existe. \\
\texttt{w+}   & Lectura y escritura. \textbf{Sobreescribe} (borra lo anterior). Crea archivo si no existe. \\
\texttt{a}    & Añadir (\textbf{Append}). Escribe al final sin borrar. Crea archivo si no existe. \\
\texttt{a+}   & Lectura y añadir. Escribe al final sin borrar. Crea archivo si no existe. \\
\bottomrule
\end{tabularx}
\end{frame}


\begin{frame}[fragile]\frametitle{Abriendo un Archivo}
\begin{block}{}
\centering Se recomienda comprobar si el archivo existe antes de abrirlo para leer
\end{block}
\begin{lstlisting}[basicstyle=\small]
fichero = open("mi_archivo.txt", mode="r")
# Hacer cosas
fichero.close() # No olvides cerrar el archivo
\end{lstlisting}
\begin{lstlisting}[basicstyle=\small]
with open("mi_archivo.txt", mode="r") as fichero:

\end{lstlisting}
\end{frame}


\begin{frame}[fragile]\frametitle{Leer el Contenido}
\begin{itemize}
\item \lstinline|file.readline() -> str|
    \begin{itemize}
    \item Leer una sola línea
    \end{itemize}
\item \lstinline|file.read() -> str|
    \begin{itemize}
        \item Leer todo el archivo y almacenarlo en string
    \end{itemize}

    \begin{itemize}

    \end{itemize}
\end{itemize}
\end{frame}

\begin{lstlisting}[basicstyle=\small]

\end{frame}



\subsection{Usando F-Strings}

\begin{frame}[fragile]\frametitle{Evolución: Formateando la Salida}
¿Cómo mezclamos texto y variables?
\begin{lstlisting}[basicstyle=\tiny]
nombre = "Ana"; asig = "Datos"

# 1. Por comas (Añade espacios automaticos)
print('Hola', nombre, '! Bienvenido a', asig)

# 2. Operador % (Antiguo, estilo C)
print('Hola %s! Bienvenido a %s.' % (nombre, asig))

# 3. Metodo .format() (Mas moderno)
print('Hola {}! Bienvenido a {}.'.format(nombre, asig))

# 4. F-Strings (RECOMENDADO: Mas rapido y legible)
print(f'Hola {nombre}! Bienvenido a {asig}.')
\end{lstlisting}
\terminal{Hola Ana ! Bienvenido a Datos\\Hola Ana! Bienvenido a Datos.\\Hola Ana! Bienvenido a Datos.\\Hola Ana! Bienvenido a Datos.}
\end{frame}

\begin{frame}[fragile]\frametitle{Usando F-Strings}
\begin{itemize}
    \item \textbf{F-String} (\lstinline|f"..."|): Permite incrustar código Python ("vivo") dentro de \lstinline|{...}|.
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
nombre = "Ana"

# 1. String Normal (Literal, tonto)
print("Hola {nombre}") 

# 2. F-String (Inteligente, evalúa variables)
print(f"Hola {nombre}")
\end{lstlisting}
\terminal{Hola \{nombre\}\\Hola Ana}

\begin{lstlisting}[basicstyle=\scriptsize]
# ¡Admite cualquier expresión!
print(f"Mañana cumpliras {25 + 1} años")
\end{lstlisting}
\terminal{Mañana cumpliras 26 años}
\end{frame}

\begin{frame}[fragile]\frametitle{Formateando F-Strings}
\begin{itemize}
\item Podemos pedirle a Python que formatee la salida de varias maneras
\item Sigue la misma \href{https://docs.python.org/3/library/string.html#format-specification-mini-language}{\color{blue}\underline{especificación}} que \lstinline|str.format()|
\item Usa dos puntos para separar el valor del formato
\end{itemize}
\begin{lstlisting}[basicstyle=\small]
numero = 123.123141241241
# El punto especifica el número de decimales
# La f al final es para tratarlo como float
print(f'Número con 3 decimales={numero:.3f}')

# Podemos especificar la dirección con <, >, ^
# Podemos reservar un espacio de 10 de ancho para el número
print(f'Más ancho y a la derecha={numero:>10.3f}')
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Formateando F-Strings, Uso Básico}
\begin{block}{Sintaxis (Mini-Lenguaje de Formato)}
\centering\verb|:[fill][align][width][.precision][type]|
\end{block}
\begin{description}
\item[fill (relleno):] Carácter para rellenar espacios huecos (ej: \lstinline|*|, \lstinline|0|).
\item[align (alineación):] \lstinline|<| (izq), \lstinline|>| (der), \lstinline|^| (centro).
\item[width (ancho):] Número mínimo de caracteres que ocupará.
\item[.precision:] Cuantos decimales mostrar (ej: \lstinline|.2f|).
\item[type:] Tipo de dato (ej: \lstinline|f| para float, \lstinline|d| para entero).
\end{description}
\begin{lstlisting}[basicstyle=\tiny]
pi = 3.14159
print(f"|{pi:*>10.2f}|")  # Relleno=*, Align=>, Ancho=10, Prec=.2, Tipo=f
\end{lstlisting}
\terminal{|******3.14|}
\end{frame}


\subsection{Anotaciones de Tipo}

\begin{frame}[fragile]\frametitle{Tipado en Python (Type Hints)}
\begin{itemize}
    \item \textbf{Tipado Dinámico:} Python decide el tipo al ejecutar el código.
    \item \textbf{Type Hints:} "Sugerencias" de tipo para humanos e IDEs.
    \item \textbf{¿Es obligatorio?} No. Python las ignora al ejecutar.
    \item \textbf{Utilidad:} Documentación clara y ayuda en Spyder/VSCode.
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
# Indicamos que 'nombre' es str y la funcion devuelve str
def saludar(nombre: str) -> str:
    return f"Hola {nombre}"

edad: int = 25 # Tambien en variables
\end{lstlisting}
\scriptsize \textit{Nota: Introducido en 3.5, mejorado en 3.9+}
\end{frame}

\begin{frame}[fragile]\frametitle{Tipado Simple}
\begin{block}{}
\centering Podemos añadir un tipo a cualquier variable añadiendo dos puntos después del nombre
\end{block}
\begin{lstlisting}[basicstyle=\scriptsize]
# Formato: nombre: tipo = valor
a: int = 5
b: float = 2.3
c: list = []

def sumar(x: int, y: int) -> int:
    return x + y
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Complicando los Tipos}
\begin{itemize}
\item Desde la versión 3.9 de Python podemos especificar el contenido de las colecciones
\item Para leer más, ve \href{https://mypy.readthedocs.io/en/stable/cheat_sheet_py3.html}{\color{blue}\underline{aquí}}
\begin{lstlisting}[basicstyle=\scriptsize]
# m: Una lista que contiene listas de enteros
m: list[list[int]] = [[1, 2], [3, 4]]

# d: Diccionario con nombres (str) y notas (float)
d: dict[str, float] = {"Ana": 9.5, "Beto": 7.0}

# Podemos ser tan especificos como queramos
# (Estructura: Clave=str, Valor=Lista de listas de ints)
a: dict[str, list[list[int]]] = {"id": [[1], [2]]}
\end{lstlisting}
\end{itemize}
\end{frame}

\subsection{Definiendo una Función \texttt{main}}

\begin{frame}[fragile]\frametitle{Buenas Prácticas: La función main}
\begin{itemize}
    \item \textbf{Problema:} Si escribimos código suelto, se ejecuta siempre y ensucia todo.
    \item \textbf{Solución:} Agrupar la lógica principal en una función \lstinline|def main():|.
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
# MAL: Código suelto (Global)
print("Arrancando...")
x = 5

# BIEN: Todo ordenado en su sitio
def main():
    print("Arrancando...")
    x = 5
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{El Guardián: if \_\_name\_\_ == "\_\_main\_\_"}
\begin{itemize}
    \item Este bloque es el \textbf{control de seguridad} de Python.
    \item Pregunta: \textit{"¿Me has dado al Play a mí (Soy el protagonista) o solo me estás importando (Soy una librería)?"}
\end{itemize}

def main():

\section{Creando y Usando una Clase}

\begin{frame}[fragile]\frametitle{¿Qué es una Clase? (El Molde)}
\begin{itemize}
    \item \textbf{Clase (El Molde):} Es el plano o plantilla. Define qué propiedades tendrá algo.
        \begin{itemize}
        \item Ej: "Un Estudiante tiene nombre y nota".
        \end{itemize}
    \item \textbf{Objeto (La Galleta):} Es la cosa real creada a partir del molde.
        \begin{itemize}
        \item Ej: "Juan (nota 5)", "Ana (nota 9)".
        \end{itemize}
\end{itemize}
\begin{block}{Analogía}
Con un solo molde de galletas (Clase) puedes hacer 100 galletas diferentes (Objetos).
\end{block}
\end{frame}

\begin{frame}[fragile]\frametitle{Partes de una Clase}
\begin{itemize}
\item Una clase tiene dos tipos de componentes:
\begin{description}
\item[Atributos (Datos):] Lo que el objeto \textbf{ES} o \textbf{TIENE}.
    \begin{itemize}
    \item Ej: \lstinline|nombre|, \lstinline|nota|, \lstinline|edad|.
    \end{itemize}
\item[Métodos (Acciones):] Lo que el objeto \textbf{HACE}.
    \begin{itemize}
    \item Ej: \lstinline|estudiar()|, \lstinline|dormir()|, \lstinline|aprobar()|.
    \end{itemize}
\end{description}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{El Nacimiento: \_\_init\_\_}
\begin{itemize}
    \item \lstinline|__init__|: Función especial (Constructor) que se ejecuta AUTOMÁTICAMENTE al crear el objeto.
    \item \lstinline|self|: Significa "YO MISMO". Permite al objeto guardar sus propios datos.
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
class Estudiante:
    def __init__(self, nombre, nota):
        # "MI nombre es igual al nombre que me pasan"
        self.nombre = nombre
        self.nota = nota

# Creando el objeto (La galleta)
# Python llama a __init__ por nosotros
juan = Estudiante("Juan", 5) 
print(juan.nombre) # Acceso directo (Publico)
\end{lstlisting}
\terminal{Juan}
\end{frame}


\begin{frame}[fragile]\frametitle{¿Por Qué el \_\_? (Privacidad)}
\begin{itemize}
    \item \textbf{Problema:} Si el dato es público, cualquiera lo puede romper.
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
juan.nota = 1000 # ¡Trampa! Esto es valido si es publico
\end{lstlisting}
\begin{itemize}
    \item \textbf{Solución:} Poner dos guiones bajos (\lstinline|__|) antes del nombre lo hace \textbf{Privado}.
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
class Estudiante:
    def __init__(self, nombre, nota):
        self.__nombre = nombre # Protegido
        self.__nota = nota     # Protegido

# juan.__nota = 1000  -> ERROR (Python no lo encuentra)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Añadiendo un Método (Acción)}
\begin{itemize}
    \item \textbf{Definición:} \lstinline|self| es OBLIGATORIO como primer parámetro.
    \item \textbf{Llamada:} \lstinline|self| es AUTOMÁTICO (Python lo pone invisiblemente).
\end{itemize}
\begin{lstlisting}[basicstyle=\scriptsize]
class Estudiante:
    def __init__(self, nombre, nota):
        self.__nombre = nombre; self.__nota = nota
        
    # DEFINICION: self es obligatorio aqui
    def estudiar(self, horas):
        self.__nota += (horas * 0.1)
        print(f"Nueva nota: {self.__nota}")

juan = Estudiante("Juan", 5)

# LLAMADA: NO pasamos self (Python pasa a 'juan' por nosotros)
juan.estudiar(10) 
\end{lstlisting}
\terminal{Nueva nota: 6.0}
\end{frame}

\begin{frame}[fragile]\frametitle{La Magia del self (¿Qué pasa por debajo?)}
\begin{itemize}
\item ¿Por qué definimos 2 parámetros (\lstinline|self, horas|) pero solo pasamos 1 (\lstinline|5|)?
\item \textbf{Respuesta:} Porque Python transforma tu código:
\end{itemize}
\begin{lstlisting}[basicstyle=\small]
# TU ESCRIBES:
juan.estudiar(5)

# PYTHON REALMENTE EJECUTA:
# Pasa el objeto 'juan' como primer argumento (self)
Estudiante.estudiar(juan, 5)
\end{lstlisting}
\begin{block}{Regla Nemotécnica}
El sujeto de la frase (`juan`) se convierte automáticamente en el `self`.
\end{block}
\end{frame}
